# --- PROMO_ACTIVATED bulk fan-out (see PromoActivatedConsumer) ---
# Same user-service base URL discount-loyalty-service already points at
# for its own (single-user) Feign client - this service's client just
# calls a paginated variant of the same /api/v1/admin/users endpoint.
user-service.url=${USER_SERVICE_URL:http://localhost:8081}
# Item 4 (proper fix): this service's secret for user-service's
# POST /internal/oauth/token, matching user-service's
# security.service-clients.clients.notification-service.secret. When set,
# ServiceTokenProvider exchanges it for a short-lived JWT and refreshes it
# automatically - user-service.service-token below is then ignored.
user-service.service-secret=${SERVICE_CLIENT_NOTIFICATION_SECRET:CHANGE_ME_NOTIFICATION_SERVICE_SECRET}
# Legacy static service-to-service credential sent as "Authorization: Bearer
# <token>" on every call UserServiceFeignClient makes (see
# UserServiceFeignConfig). Kept only as a fallback for as long as
# user-service.service-secret above is not set; left blank only for
# local/dev convenience, in which case a warning is logged and the call
# goes out unauthenticated.
user-service.service-token=${USER_SERVICE_SERVICE_TOKEN:}
# Sent as X-Client-ID on the same calls. user-service's ClientIdValidationFilter
# runs BEFORE its JWT filter and rejects /api/v1/admin/** without an approved
# client id with 400 USR-400-001, so the Authorization header alone was never
# enough. This value must also appear in user-service's
# security.client.approved-ids list.
user-service.client-id=${USER_SERVICE_CLIENT_ID:notification-service}
# Users resolved/notified per page of the eligible-user pagination. Keeps
# each page bounded and restartable instead of loading the whole 50,000+
# audience into memory or one Kafka message at once.
notification.bulk.batch-size=${NOTIFICATION_BULK_BATCH_SIZE:1000}
# Safety cap on pages processed per campaign poll, purely to bound a
# runaway loop if user-service ever returns "last=false" forever; not
# expected to be hit in normal operation.
notification.bulk.max-pages-per-campaign=${NOTIFICATION_BULK_MAX_PAGES_PER_CAMPAIGN:1000}
# How long a campaign can sit in IN_PROGRESS with no update before
# PromoCampaignRecoveryScheduler treats it as stalled (e.g. paused after
# hitting max-pages-per-campaign, or abandoned after a crash) and resumes
# it from lastProcessedPage + 1.
notification.bulk.stale-campaign-threshold-minutes=${NOTIFICATION_BULK_STALE_CAMPAIGN_THRESHOLD_MINUTES:15}
# How often PromoCampaignRecoveryScheduler polls for stalled campaigns.
notification.bulk.recovery-poll-interval-ms=${NOTIFICATION_BULK_RECOVERY_POLL_INTERVAL_MS:60000}
# Geofence alerts go to the operator; set this to also copy a platform address.
notification.alerts.platform-email=${NOTIFICATION_PLATFORM_ALERT_EMAIL:}
notification.alerts.geofence-topic=${NOTIFICATION_GEOFENCE_TOPIC:tracking.geofence.event}
